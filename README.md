инструкция по запуску
# 1. Клонируйте репозиторий
git clone https://github.com/AlexandraSimaeva/Calculator.git
cd Calculator

# 2. Соберите проект через Сmake
mkdir -p build
cd build
cmake ..
make
# 3. Запустите сервер
./build/Calculator  
# Пример теста( у меня порты 8080, 8088 не работают, поэтому использовала 9090)
curl -X POST http://localhost:9090 \
     -H "Content-Type: application/json" \
     -d '{"exp":"2+2*3"}''
